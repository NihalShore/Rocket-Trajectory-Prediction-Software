cmake_minimum_required(VERSION 3.16)
project(rocket_trajectory_core VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(rocket_core src/atmosphere.cpp src/vehicle.cpp src/integrator.cpp)
target_include_directories(rocket_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(rocket_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(rocket_demo examples/demo.cpp)
target_link_libraries(rocket_demo PRIVATE rocket_core)
enable_testing()
add_executable(rocket_core_tests tests/core_tests.cpp)
target_link_libraries(rocket_core_tests PRIVATE rocket_core)
add_test(NAME rocket_core_tests COMMAND rocket_core_tests)
